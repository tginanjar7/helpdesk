  function setRating(score) {
    const container = document.getElementById('star-rating-box');

    const labels = [
      '',
      '1.0 / 5.0 (Sangat Tidak Puas)',

      '3.0 / 5.0 (Cukup)',
      '4.0 / 5.0 (Puas)',

    buttons.forEach((btn, index) => {

      if (index
      score) {
        star.style.fontVariationSettings = "'FILL' 1";
        btn.classList.add('text-surface-tint');

      } else {

        btn.classList.remove('text-surface-tint');
        btn.classList.add('text-outline-variant');
      }
    });
    document.getElementById('rating-label').innerText = labels[score];
  }
